const scales = [
  "d#:minor"
]

let scaleSelector = choose(...scales).seg(1)

const bassModifier = [
  "0 0 0 -7 0 0 1 -7",
  "0 0 0 -7 0 0 0 4",
  "0 0 0 -7 0 0 -7 -7"
]

const arpArray = [
  "<0 2 4 2 8>*16",
  "<0 5 7 9 8>*16",
  "<0 3 6 8 2>*16",
  "<6 4 5 7 9>*16",
]

CYBERBASS: note("0 0 0 0 0 0 2 1".add(choose(...bassModifier).seg(2))).s("z_sawtooth")
  .slow(4)
  .scale(scaleSelector)
  .vib(0.5)
  .vibmod(0.6)
  .trans(-12)
  .decay(0.4)
  .sustain(1)
  .release(0.4)
  .acidenv(slider(0.28,0,1,0.02))
  .distort("2:0.22:diode")
  .delay(slider(0.54,0,1,0.02))
  .delayspeed(0.3)
  .room(1)
  .rsize(1.5)
  .gain(slider(0.88,0,1,0.02))
  .o(2)
  .scope()

ARPLEAD: note(pick(arpArray, "<0 1 2 3>").slow(2)).sound("supersaw")
  .trans(choose(12, 0, -12).slow(4))
  .scale(scaleSelector)
  ,legato()
  .delay(0.33)
  .delayspeed(0.3)
  .room(1)
  .rsize(1.5)
  .gain(slider(0.35,0,1,0.05))
  ._pianoroll()

KICK: s(`
  [bd bd bd bd]
`).bank("akaimpc60")
  .lpf(600)
  .duckorbit(2)
  .gain(slider(1,0,1,0.02))
  .distort("1.2:0.5:diode")
  ._punchcard()
HATS: s(`
  [hh - hh*2 - hh - hh - hh - hh*3 - hh*4 - hh -]
`).bank("akaimpc60")
  .hpf(sine.range(6000,9000).fast(8))
  .lpf(9100).lpq(4)
  .room(0.5).rsize(0.6)
  .gain(slider(0.08,0,1,0.02))
  ._punchcard()
TOP: s(`
  [- hh - hh - hh - hh ]
`).bank("mc303")
  .room(0.6).rsize(0.9)
  .postgain(slider(0.06,0,1,0.02))
  ._punchcard()
